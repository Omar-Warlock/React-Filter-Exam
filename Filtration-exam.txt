		5-    class BankAccount {
  constructor(accountNumber, balance = 0) {
    this.accountNumber = accountNumber;
    this.balance = balance;
  }

  deposit(amount) {
    if (amount > 0) {
      this.balance += amount;
      console.log(`Deposited $${amount} to account ${this.accountNumber}`);
    } else {
      console.log("Deposit amount must be positive.");
    }
  }

  withdraw(amount) {
    if (amount > 0 && amount <= this.balance) {
      this.balance -= amount;
      console.log(`Withdrew $${amount} from account ${this.accountNumber}`);
    } else {
      console.log("Insufficient balance or invalid amount.");
    }
  }

  getBalance() {
    console.log(`Balance for account ${this.accountNumber}: $${this.balance}`);
  }
}

const account1 = new BankAccount("1");
const account2 = new BankAccount("2", 1000);

account1.deposit(500);     // Deposited $500
account1.withdraw(200);    // Withdrew $200
account1.getBalance();     // Balance: $300

account2.withdraw(300);    // Withdrew $300
account2.deposit(150);     // Deposited $150
account2.getBalance();     // Balance: $850

function binarySearch(arr,goal){
        
        let low = 0 ;
        let high = arr.length -1 ;
        
        
        while(low <= high){
            const middle = Math.floor((low + high) / 2)
        
            if(arr[middle] === goal){
                return middle
            }
            
            if(arr[middle] > goal){
                high = middle -1;
            }else{
                low = middle + 1
            }
        }
        
        return -1
}

import React, { useRef, useState } from "react";

function Dropdown() {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef(null);

  const toggleDropdown = () => setIsOpen(!isOpen);

  return (
    <div ref={dropdownRef}>
      <button onClick={toggleDropdown}>Toggle Menu</button>
      {isOpen && (
        <ul>
          <li>Option 1</li>
          <li>Option 2</li>
        </ul>
      )}
    </div>
  );
}
